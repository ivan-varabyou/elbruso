
-- 1. Языки
CREATE TABLE languages (
    id SERIAL PRIMARY KEY,
    code VARCHAR(5) NOT NULL UNIQUE, -- 'ru', 'en'
    name VARCHAR(50) NOT NULL,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- 2. Страны
CREATE TABLE countries (
    id SERIAL PRIMARY KEY,
    code_alpha2 VARCHAR(2) NOT NULL UNIQUE,      -- RU, BY, KZ
    code_alpha3 VARCHAR(3) NOT NULL UNIQUE,      -- RUS, BLR, KAZ
    name_ru VARCHAR(100) NOT NULL,               -- Эталонное название на русском
    capital_city_code VARCHAR(10),               -- Код столицы
    is_active BOOLEAN DEFAULT TRUE,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- 3. Федеральные округа
CREATE TABLE federal_districts (
    id SERIAL PRIMARY KEY,
    country_id INTEGER NOT NULL REFERENCES countries(id) ON DELETE CASCADE,
    code VARCHAR(10) NOT NULL UNIQUE, -- CFO, SZO, YFO
    name_ru VARCHAR(100) NOT NULL,    -- Эталонное название на русском
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- 4. Типы регионов
CREATE TABLE region_types (
    id SERIAL PRIMARY KEY,
    code VARCHAR(20) NOT NULL UNIQUE,
    name_ru VARCHAR(50) NOT NULL,     -- Эталонное название на русском
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- 5. Регионы 
CREATE TABLE regions (
    id SERIAL PRIMARY KEY,
    country_id INTEGER NOT NULL REFERENCES countries(id) ON DELETE CASCADE,
    federal_district_id INTEGER REFERENCES federal_districts(id) ON DELETE SET NULL,
    region_type_id INTEGER REFERENCES region_types(id),
    code VARCHAR(10) NOT NULL,        -- MOW, SPE, MOS
    name_ru VARCHAR(100) NOT NULL,    -- Эталонное название на русском
    is_active BOOLEAN DEFAULT TRUE,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    UNIQUE(country_id, code)
);

-- 6. Типы организаций
CREATE TABLE organization_levels (
    id SERIAL PRIMARY KEY,
    code VARCHAR(20) NOT NULL UNIQUE, -- INTERNATIONAL, NATIONAL, CLUB
    name_ru VARCHAR(50) NOT NULL,     -- Эталонное название на русском
    sort_order INTEGER NOT NULL DEFAULT 0,
    is_active BOOLEAN DEFAULT TRUE,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- 7. Организации
CREATE TABLE organizations (
    id SERIAL PRIMARY KEY,
    internal_code VARCHAR(50) UNIQUE, -- ORG_INT_IOC, ORG_NAT_RU_FOOTBALL
    level_id INTEGER NOT NULL REFERENCES organization_levels(id),
    name_ru VARCHAR(200) NOT NULL,    -- Эталонное название на русском
    parent_id INTEGER REFERENCES organizations(id) ON DELETE SET NULL,
    metadata JSONB DEFAULT '{}'::jsonb,
    is_active BOOLEAN DEFAULT TRUE,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    CONSTRAINT no_self_parent CHECK (id != parent_id)
);

-- 8. Таблица переводов
CREATE TABLE translations (
    id SERIAL PRIMARY KEY,
    entity_type VARCHAR(50) NOT NULL, -- 'country', 'region', 'organization', 'federal_district'
    entity_id INTEGER NOT NULL,
    language_code VARCHAR(5) NOT NULL, -- 'en', 'fr', 'de', etc.
    field_name VARCHAR(50) NOT NULL, -- 'name', 'description', 'abbreviation'
    translation TEXT NOT NULL,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    UNIQUE(entity_type, entity_id, language_code, field_name)
);

-- Таблица категорий Олимпийских игр
CREATE TABLE olympic_categories (
    id SERIAL PRIMARY KEY,
    code VARCHAR(20) NOT NULL UNIQUE,  -- summer, winter, both, none
    name_ru VARCHAR(50) NOT NULL,      -- Летние, Зимние, Оба, Неолимпийский
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- Таблица типов спорта (командный/индивидуальный)
CREATE TABLE sport_types (
    id SERIAL PRIMARY KEY,
    code VARCHAR(20) NOT NULL UNIQUE,  -- team, individual, mixed
    name_ru VARCHAR(50) NOT NULL,      -- Командный, Индивидуальный, Смешанный
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- Основная таблица видов спорта
CREATE TABLE sports (
    id SERIAL PRIMARY KEY,
    code VARCHAR(50) NOT NULL UNIQUE,      -- football, basketball, athletics
    name_ru VARCHAR(100) NOT NULL,         -- Название на русском (эталонное)
    olympic_category_id INTEGER REFERENCES olympic_categories(id),
    sport_type_id INTEGER REFERENCES sport_types(id),
    popularity_score INTEGER DEFAULT 0,    -- Оценка популярности (0-100)
    is_active BOOLEAN DEFAULT TRUE,
    metadata JSONB DEFAULT '{}',           -- Дополнительные данные
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);
